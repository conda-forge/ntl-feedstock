# CMake build for NTL on Windows (MSVC).
#
# NTL's own build system assumes a Unix compiler driver (-o, -l, .o objects, ar)
# and cannot drive MSVC directly, so this file reimplements what ./configure and
# the setup phase of the makefile do (see src/DoConfig and src/mfile), following
# what upstream does for its own Windows package (see "make winpack" and
# src/dosify) where the two differ.

cmake_minimum_required(VERSION 3.25)
project(ntl CXX)

option(NTL_GMP_LIP "Use GMP for long integer arithmetic" ON)
option(NTL_GF2X_LIB "Use gf2x for GF2X multiplication" ON)
include(CTest)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

set(NTL_SRC ${CMAKE_CURRENT_SOURCE_DIR}/src)
set(NTL_GENERATED ${CMAKE_CURRENT_BINARY_DIR}/include)
set(NTL_INCLUDE_DIRS ${NTL_GENERATED} ${CMAKE_CURRENT_SOURCE_DIR}/include)
file(MAKE_DIRECTORY ${NTL_GENERATED}/NTL)

if(NTL_GMP_LIP)
  find_path(GMP_INCLUDE_DIR NAMES gmp.h REQUIRED)
  find_library(GMP_LIBRARY NAMES gmp REQUIRED)
  list(APPEND NTL_INCLUDE_DIRS ${GMP_INCLUDE_DIR})
endif()
if(NTL_GF2X_LIB)
  find_path(GF2X_INCLUDE_DIR NAMES gf2x.h REQUIRED)
  find_library(GF2X_LIBRARY NAMES gf2x REQUIRED)
  list(APPEND NTL_INCLUDE_DIRS ${GF2X_INCLUDE_DIR})
endif()

# config.h: substitute the @{...} placeholders in src/cfile like DoConfig does.
# Values are DoConfig's defaults except where noted.
set(NTL_CONFIG_ON
  NTL_STD_CXX11
  NTL_DISABLE_MOVE_ASSIGN
  NTL_NO_INIT_TRANS
  NTL_CLEAN_PTR
  NTL_SAFE_VECTORS
)
# NTL_THREADS (and with it NTL_THREAD_BOOST and NTL_TLS_HACK) is off like in
# build.sh.
if(NTL_GMP_LIP)
  list(APPEND NTL_CONFIG_ON NTL_GMP_LIP)
endif()
if(NTL_GF2X_LIB)
  list(APPEND NTL_CONFIG_ON NTL_GF2X_LIB)
endif()

file(READ ${NTL_SRC}/cfile NTL_CONFIG)
string(REGEX MATCHALL "@{[A-Z0-9_]+}" NTL_CONFIG_PLACEHOLDERS "${NTL_CONFIG}")
list(REMOVE_DUPLICATES NTL_CONFIG_PLACEHOLDERS)
foreach(placeholder IN LISTS NTL_CONFIG_PLACEHOLDERS)
  string(REGEX REPLACE "^@{(.*)}$" "\\1" name ${placeholder})
  if(name STREQUAL "NTL_UNSIGNED_LONG_LONG_TYPE")
    set(value "unsigned long long")
  elseif(name STREQUAL "WIZARD_HACK")
    set(value "")
  elseif(name IN_LIST NTL_CONFIG_ON)
    set(value 1)
  else()
    set(value 0)
  endif()
  string(REPLACE "${placeholder}" "${value}" NTL_CONFIG "${NTL_CONFIG}")
endforeach()
file(WRITE ${NTL_GENERATED}/NTL/config.h "${NTL_CONFIG}")

# mach_desc.h: upstream ships a pregenerated version for Windows.
configure_file(${NTL_SRC}/mach_desc.win ${NTL_GENERATED}/NTL/mach_desc.h COPYONLY)

# HAVE_*.h: run the feature checks like src/MakeCheckFeatures. Checks for
# instruction set extensions are skipped since MSVC compiles intrinsics without
# special flags, so these checks would enable them depending on the CPU of the
# build machine.
set(NTL_FEATURES
  ALIGNED_ARRAY BUILTIN_CLZL LL_TYPE SSSE3 AVX PCLMUL AVX2 FMA AVX512F
  COPY_TRAITS1 COPY_TRAITS2 CHRONO_TIME MACOS_TIME POSIX_TIME AES_NI KMA
)
set(NTL_ISA_FEATURES SSSE3 AVX PCLMUL AVX2 FMA AVX512F AES_NI KMA)
foreach(feature IN LISTS NTL_FEATURES)
  file(WRITE ${NTL_GENERATED}/NTL/HAVE_${feature}.h "\n")
endforeach()
foreach(feature IN LISTS NTL_FEATURES)
  if(feature IN_LIST NTL_ISA_FEATURES)
    continue()
  endif()
  try_run(NTL_RUN_${feature} NTL_COMPILE_${feature}
    SOURCES ${NTL_SRC}/Check${feature}.cpp ${NTL_SRC}/ctools.cpp
    COMPILE_DEFINITIONS -DNTL_FCHECK
    CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${NTL_GENERATED};${CMAKE_CURRENT_SOURCE_DIR}/include"
  )
  if(NTL_COMPILE_${feature} AND NTL_RUN_${feature} EQUAL 0)
    message(STATUS "Checking for feature ${feature} - yes")
    file(WRITE ${NTL_GENERATED}/NTL/HAVE_${feature}.h
      "#ifndef NTL_HAVE_${feature}\n#define NTL_HAVE_${feature}\n#endif\n")
  else()
    message(STATUS "Checking for feature ${feature} - no")
  endif()
endforeach()

# gmp_aux.h: generated by gen_gmp_aux like setup3 in src/mfile.
if(NTL_GMP_LIP)
  try_run(NTL_RUN_GEN_GMP_AUX NTL_COMPILE_GEN_GMP_AUX
    SOURCES ${NTL_SRC}/gen_gmp_aux.cpp
    CMAKE_FLAGS "-DINCLUDE_DIRECTORIES=${NTL_INCLUDE_DIRS}"
    LINK_LIBRARIES ${GMP_LIBRARY}
    COMPILE_OUTPUT_VARIABLE NTL_GEN_GMP_AUX_COMPILE_OUTPUT
    RUN_OUTPUT_STDOUT_VARIABLE NTL_GMP_AUX
    RUN_OUTPUT_STDERR_VARIABLE NTL_GEN_GMP_AUX_STDERR
  )
  if(NOT NTL_COMPILE_GEN_GMP_AUX)
    message(FATAL_ERROR "Could not build gen_gmp_aux:\n${NTL_GEN_GMP_AUX_COMPILE_OUTPUT}")
  endif()
  if(NOT NTL_RUN_GEN_GMP_AUX EQUAL 0)
    message(FATAL_ERROR "gen_gmp_aux failed:\n${NTL_GEN_GMP_AUX_STDERR}")
  endif()
  file(WRITE ${NTL_GENERATED}/NTL/gmp_aux.h "${NTL_GMP_AUX}")
endif()

# The library sources are the ones listed as SRC in the makefile. GetTime and
# GetPID are chosen like in upstream's Windows package.
file(READ ${NTL_SRC}/mfile NTL_MAKEFILE)
string(REGEX MATCH "\nSRC=([^\n]*\\\\\n)*[^\n]*" NTL_SOURCES "${NTL_MAKEFILE}")
string(REGEX REPLACE "^\nSRC=" "" NTL_SOURCES "${NTL_SOURCES}")
string(REGEX REPLACE "[ \t\\\\\n]+" ";" NTL_SOURCES "${NTL_SOURCES}")
list(FILTER NTL_SOURCES EXCLUDE REGEX "^$")
list(TRANSFORM NTL_SOURCES PREPEND ${NTL_SRC}/)
list(APPEND NTL_SOURCES ${NTL_SRC}/GetTime4.cpp ${NTL_SRC}/GetPID2.cpp)

# NTL can only be built as a static library with MSVC: its headers access
# global data from inline functions, e.g., ZZ_p::modulus(), which cannot be
# imported from a DLL without __declspec(dllimport) annotations.
add_library(ntl STATIC ${NTL_SOURCES})
target_include_directories(ntl PUBLIC
  "$<BUILD_INTERFACE:${NTL_INCLUDE_DIRS}>"
  $<INSTALL_INTERFACE:include>
)
target_link_libraries(ntl PUBLIC
  $<$<BOOL:${NTL_GMP_LIP}>:${GMP_LIBRARY}>
  $<$<BOOL:${NTL_GF2X_LIB}>:${GF2X_LIBRARY}>
)

install(TARGETS ntl ARCHIVE DESTINATION lib)
file(GLOB NTL_HEADERS ${CMAKE_CURRENT_SOURCE_DIR}/include/NTL/*.h)
file(GLOB NTL_GENERATED_HEADERS ${NTL_GENERATED}/NTL/*.h)
install(FILES ${NTL_HEADERS} ${NTL_GENERATED_HEADERS} DESTINATION include/NTL)

# The tests run by "make check".
if(BUILD_TESTING)
  set(NTL_TESTS
    QuickTest ZZTest SSMulTest ZZ_pXTest lzz_pXTest MoreFacTest GF2XTest
    GF2EXTest GF2EXGCDTest mat_lzz_pTest BitMatTest ZZ_pEXTest ZZ_pEXGCDTest
    lzz_pEXTest lzz_pEXGCDTest ThreadTest
  )
  # These are checked against the expected output in src/*TestOut.
  set(NTL_TESTS_WITH_OUTPUT
    CanZassTest BerlekampTest ZZXFacTest MatrixTest CharPolyTest RRTest
    QuadTest LLLTest
  )
  foreach(test IN LISTS NTL_TESTS NTL_TESTS_WITH_OUTPUT)
    add_executable(${test} ${NTL_SRC}/${test}.cpp)
    target_link_libraries(${test} PRIVATE ntl)
  endforeach()
  foreach(test IN LISTS NTL_TESTS)
    if(EXISTS ${NTL_SRC}/${test}In)
      add_test(NAME ${test}
        COMMAND ${CMAKE_COMMAND} "-DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}"
                -DTEST=$<TARGET_FILE:${test}> -DINPUT=${NTL_SRC}/${test}In
                -P ${CMAKE_CURRENT_SOURCE_DIR}/RunTest.cmake)
    else()
      add_test(NAME ${test} COMMAND ${test})
    endif()
  endforeach()
  foreach(test IN LISTS NTL_TESTS_WITH_OUTPUT)
    add_test(NAME ${test}
      COMMAND ${CMAKE_COMMAND} "-DEMULATOR=${CMAKE_CROSSCOMPILING_EMULATOR}"
              -DTEST=$<TARGET_FILE:${test}> -DINPUT=${NTL_SRC}/${test}In
              -DEXPECTED=${NTL_SRC}/${test}Out -P ${CMAKE_CURRENT_SOURCE_DIR}/RunTest.cmake)
  endforeach()
endif()
